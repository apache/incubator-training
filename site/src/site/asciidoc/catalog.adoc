<script>
(function() {
  var catalog = null;
  var activeTags = new Set();

  function loadCatalog() {
    var xhr = new XMLHttpRequest();
    xhr.open('GET', 'catalog.json', true);
    xhr.onload = function() {
      if (xhr.status === 200) {
        catalog = JSON.parse(xhr.responseText);
        renderTags();
        renderModules();
      } else {
        document.getElementById('catalogGrid').innerHTML =
          '<div class="no-results">Could not load catalog. Run <code>python3 tools/generate-catalog.py</code> first.</div>';
      }
    };
    xhr.onerror = function() {
      document.getElementById('catalogGrid').innerHTML =
        '<div class="no-results">Could not load catalog. Run <code>python3 tools/generate-catalog.py</code> first.</div>';
    };
    xhr.send();
  }

  function renderTags() {
    var container = document.getElementById('tagFilters');
    container.innerHTML = '';
    catalog.tags.forEach(function(tag) {
      var chip = document.createElement('span');
      chip.className = 'tag-chip';
      chip.textContent = tag;
      chip.addEventListener('click', function() {
        if (activeTags.has(tag)) {
          activeTags.delete(tag);
          chip.classList.remove('active');
        } else {
          activeTags.add(tag);
          chip.classList.add('active');
        }
        renderModules();
      });
      container.appendChild(chip);
    });
  }

  function renderModules() {
    var grid = document.getElementById('catalogGrid');
    var countEl = document.getElementById('catalogCount');
    var searchTerm = (document.getElementById('catalogSearch').value || '').toLowerCase();

    var filtered = catalog.modules.filter(function(mod) {
      // Tag filter: module must have ALL active tags
      if (activeTags.size > 0) {
        var moduleTags = mod.tags || [];
        var hasAll = true;
        activeTags.forEach(function(t) {
          if (moduleTags.indexOf(t) === -1) hasAll = false;
        });
        if (!hasAll) return false;
      }
      // Text search
      if (searchTerm) {
        var haystack = (mod.title + ' ' + mod.description + ' ' + (mod.tags || []).join(' ')).toLowerCase();
        if (haystack.indexOf(searchTerm) === -1) return false;
      }
      return true;
    });

    countEl.textContent = filtered.length + ' of ' + catalog.count + ' modules';

    if (filtered.length === 0) {
      grid.innerHTML = '<div class="no-results">No modules match your filters. Try removing some tags.</div>';
      return;
    }

    grid.innerHTML = filtered.map(function(mod) {
      var meta = [];
      if (mod.level) meta.push(mod.level);
      if (mod.duration) meta.push(mod.duration);
      if (mod.language) meta.push(mod.language);
      if (mod.slides) meta.push(mod.slides + ' slides');

      var tagsHtml = (mod.tags || []).map(function(t) {
        return '<span>' + t + '</span>';
      }).join('');

      return '<div class="module-card">' +
        '<h3><a href="' + mod.path + '" target="_blank">' + mod.title + '</a></h3>' +
        '<div class="description">' + mod.description + '</div>' +
        (meta.length ? '<div class="meta">' + meta.join(' · ') + '</div>' : '') +
        '<div class="tags">' + tagsHtml + '</div>' +
        '</div>';
    }).join('');
  }

  // Search handler
  document.getElementById('catalogSearch').addEventListener('input', function() {
    if (catalog) renderModules();
  });

  // Load on page ready
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', loadCatalog);
  } else {
    loadCatalog();
  }
})();
</script>
